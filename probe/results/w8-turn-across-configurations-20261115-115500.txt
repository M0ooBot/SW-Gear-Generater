# --- the script (a scratch file beside the run; not kept in the repo) ---
"""Does a turn given in one configuration show in the others? On run 1's copy of s1.master (this run's own folder),
the turn made two ways, each read in every configuration through fresh handles after a show and a top-only rebuild:
(A) W5's drive by the angle mate's own value in 2H only (SetSystemValue3(…, 3, ["2H"])), the mate then deleted;
(B) after putting everything back, W5's drive by a global "Drive" linked to the angle mate's D1, the mate deleted.
Closed without saving at the end."""
import sys, os, math, json
sys.path.insert(0, sys.argv[1])
from gear_generator import swcom, store, swbuild
from gear_generator.gears import equations
pid, folder = int(sys.argv[2]), sys.argv[3]
c = [x for x in swcom.find_candidates() if x.pid == pid][0]
s = swcom.Session(c.app, c.revision, c.pid)
rec = store.read_link_record(os.path.join(folder, "TransferCase_E4.xlsx"))
box = rec.gearbox
asm = box.assembly
bodies = [(x.gear, x.name) for x in box.components] + [("Carrier carrier", cc.component) for cc in box.carriers]
sun = next(n for g, n in bodies if g == "Sun")
TRAIN = swcom.ASSEMBLY_TRAIN

def swbuild_gap(a, b):
    return max([abs(float(x) - float(y)) for x, y in zip(a[1], b[1])] + [abs(float(x) - float(y)) for x, y in zip(a[0], b[0])])


def turn(a, b):
    t = swbuild.turn_between(a, b)
    return None if t is None else round(math.degrees(t), 6)

def frames(mode):
    s.show_configuration(mode); s.rebuild(kind=TRAIN)
    out = {}
    for i, (g, n) in enumerate(bodies):
        s.find_component(f"f{i}", n); out[g] = s.component_frame(f"f{i}")
    return out

def all_frames():
    return {m: frames(m) for m in box.modes}

def report(label, base, now):
    for m in box.modes:
        turns = {g: turn(base[m][g][0], now[m][g][0]) for g, _ in bodies}
        moved = {g: t for g, t in turns.items() if t is None or abs(t) > 1e-6}
        print(f"{label}: {m}: turned from the start {moved or 'nothing'}")

BUILT_SUN = next((tuple(x.rotation), tuple(x.at)) for x in box.components if x.gear == "Sun")


def drive(route, degrees):
    s.show_configuration("2H"); s.rebuild(kind=TRAIN)
    s.find_component("dsun", sun); s.find_component("dlay", box.layout_component)
    before = {}
    for i, (g, n) in enumerate(bodies):
        s.find_component(f"b{i}", n); before[g] = s.component_frame(f"b{i}")
    now = turn(BUILT_SUN[0], before["Sun"][0])          # the Sun's turn from its built frame (θ 0 there), in 2H
    at = 90.0 + now
    if route == "global":
        s.add_global_variable("Drive", f"{at:.9f}")
    name, dim = s.add_mate("ddrive", "angle", "dsun:plane2", "dlay:plane3", at, "Repro drive")
    if route == "global":
        s.link_dimension(f"{dim}@{name}", equations.quote("Drive"))
    s.rebuild(kind=TRAIN)
    made = {}
    for i, (g, n) in enumerate(bodies):
        s.find_component(f"m{i}", n); made[g] = s.component_frame(f"m{i}")
    moved = max(swbuild_gap(before[g], made[g]) for g, _ in bodies)
    print(f"drive {route}: the Sun at {now:+.6f}° in 2H; the angle mate made at {at:.6f}°: moved {moved:.1e} on making")
    if route == "global":
        s.set_global_variable("Drive", f"{at + degrees:.9f}")
    else:
        s.set_dimension_in("ddrive", math.radians(at + degrees) * 1000.0, ["2H"])
    s.rebuild(kind=TRAIN)
    feature = s._mate_feature(name)
    doc = s._doc()
    swcom.call(doc, "ClearSelection2", True); swcom.call(feature, "Select2", False, 0)
    swcom._guarded("EditDelete", lambda: doc.EditDelete()); swcom.call(doc, "ClearSelection2", True)
    if route == "global":
        texts = [t for t, _ in s.equations()]
        for i in range(len(texts) - 1, -1, -1):
            if equations.quote("Drive") in texts[i]:
                s.delete_equation(i)
    s.rebuild(kind=TRAIN)
    print(f"drive {route}: {name} made, D1 {dim!r}, driven {degrees}° in 2H, deleted; equations now {[t for t, _ in s.equations()]}")

s.begin_build()
try:
    s.open_document(asm)
    print("active on open", s.active_configuration(), "configurations", s.configurations())
    base = all_frames()
    drive("value", 37.0)
    after_a = all_frames()
    report("A, the value route", base, after_a)
    # put every body back in every mode? Read again from here instead: B starts from where A left things.
    drive("global", 20.0)
    after_b = all_frames()
    report("B, the global route (from A's end)", after_a, after_b)
finally:
    s.end_build()
    for kind in (2, 1):
        d = swcom.call(c.app, "GetFirstDocument")
        docs = []
        while d is not None:
            docs.append(d); d = swcom.call(d, "GetNext")
        for d in docs:
            p = str(swcom.call(d, "GetPathName") or "")
            if p and os.path.normcase(p).startswith(os.path.normcase(folder)) and int(swcom.call(d, "GetType")) == kind:
                swcom.call(c.app, "QuitDoc", str(swcom.call(d, "GetTitle")))
    print("closed without saving")
